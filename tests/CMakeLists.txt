cmake_minimum_required(VERSION 3.20)
project(Tab5PDAHostTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

get_filename_component(PDA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(COMMON_ROOT "${PDA_ROOT}/lib/PDACommon" ABSOLUTE)
set(LVGL_ROOT "${PDA_ROOT}/.pio/libdeps/tab5/lvgl")
if(NOT EXISTS "${LVGL_ROOT}/src/lvgl.h")
    message(FATAL_ERROR "Run pio pkg install -e tab5 first")
endif()

add_library(pda_core
    ${COMMON_ROOT}/src/model.cpp ${COMMON_ROOT}/src/calendar.cpp
    ${COMMON_ROOT}/src/snapshot_store.cpp ${COMMON_ROOT}/src/service.cpp ${COMMON_ROOT}/src/organizer.cpp ${COMMON_ROOT}/src/backup.cpp)
target_include_directories(pda_core PUBLIC ${COMMON_ROOT}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(pda_core PUBLIC PDA_TARGET_TAB5)
target_compile_options(pda_core PRIVATE -Wall -Wextra -Wpedantic)

file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_ROOT}/src/*.c")
add_library(lvgl_host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_host SYSTEM PUBLIC ${LVGL_ROOT})
target_include_directories(lvgl_host PUBLIC ${PDA_ROOT}/include)
target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_library(pda_ui ${PDA_ROOT}/src/presentation.cpp ${PDA_ROOT}/src/ui/tools_ui.cpp ${PDA_ROOT}/src/ui/pda_font_16_bold.c ${PDA_ROOT}/src/ui/pda_font_28.c ${PDA_ROOT}/src/ui/pda_font_48.c ${PDA_ROOT}/src/productivity.cpp ${PDA_ROOT}/src/ui/productivity_ui.cpp ${PDA_ROOT}/src/workspace.cpp ${PDA_ROOT}/src/exchange.cpp ${PDA_ROOT}/src/schedule.cpp ${PDA_ROOT}/src/ui/priority_ui.cpp ${PDA_ROOT}/src/ui/extras_ui.cpp ${PDA_ROOT}/src/ui/tab_ui.cpp ${PDA_ROOT}/src/ui/pda_font_16.c)
target_include_directories(pda_ui PUBLIC ${PDA_ROOT}/include)
target_link_libraries(pda_ui PUBLIC pda_core lvgl_host)
target_compile_options(pda_ui PRIVATE -Wall -Wextra -Wpedantic)

add_executable(core_tests core_tests.cpp)
target_link_libraries(core_tests PRIVATE pda_core)
add_test(NAME core_recovery_and_rules COMMAND core_tests)
add_executable(organizer_tests organizer_tests.cpp)
target_link_libraries(organizer_tests PRIVATE pda_core)
add_test(NAME organizer_migration_alarms_and_backup COMMAND organizer_tests ${CMAKE_CURRENT_BINARY_DIR}/test-backup.jsonl)
set_tests_properties(organizer_migration_alarms_and_backup PROPERTIES FIXTURES_SETUP backup_fixture)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME usb_backup_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_usb_backup.py ${CMAKE_CURRENT_BINARY_DIR}/test-backup.jsonl)
set_tests_properties(usb_backup_protocol PROPERTIES FIXTURES_REQUIRED backup_fixture)

add_executable(ui_tests touch_tests.cpp)
target_link_libraries(ui_tests PRIVATE pda_ui)
add_test(NAME lvgl_touch_and_full_forms COMMAND ui_tests ${CMAKE_CURRENT_BINARY_DIR}/previews)
set_tests_properties(lvgl_touch_and_full_forms PROPERTIES TIMEOUT 120)

add_executable(priority_tests priority_tests.cpp)
target_include_directories(priority_tests PRIVATE ${PDA_ROOT}/include)
target_link_libraries(priority_tests PRIVATE pda_ui)
add_test(NAME priority_features_and_capacity COMMAND priority_tests)

add_test(NAME assistant_http_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_assistant_bridge.py ${CMAKE_CURRENT_BINARY_DIR}/assistant-proposal.jsonl)
set_tests_properties(assistant_http_bridge PROPERTIES FIXTURES_SETUP assistant_fixture)
add_executable(workspace_tests workspace_tests.cpp)
target_include_directories(workspace_tests PRIVATE ${PDA_ROOT}/include)
target_link_libraries(workspace_tests PRIVATE pda_ui)
add_test(NAME workspace_notes_ink_sync COMMAND workspace_tests ${CMAKE_CURRENT_BINARY_DIR}/assistant-proposal.jsonl)
set_tests_properties(workspace_notes_ink_sync PROPERTIES FIXTURES_REQUIRED assistant_fixture)

set(PDA_JS_PYTHON "" CACHE FILEPATH "Optional Python interpreter with quickjs installed")
if(PDA_JS_PYTHON)
    add_test(NAME transfer_page_javascript COMMAND ${PDA_JS_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/test_transfer_page.py)
endif()

add_executable(productivity_tests productivity_tests.cpp)
target_include_directories(productivity_tests PRIVATE ${PDA_ROOT}/include)
target_link_libraries(productivity_tests PRIVATE pda_ui)
add_test(NAME productivity_sheet_reader_finance COMMAND productivity_tests)

add_executable(robustness_tests robustness_tests.cpp)
target_link_libraries(robustness_tests PRIVATE pda_ui)
add_test(NAME malformed_inputs_and_boundaries COMMAND robustness_tests)
set_tests_properties(malformed_inputs_and_boundaries PROPERTIES TIMEOUT 30)

add_executable(presentation_tests presentation_tests.cpp)
target_link_libraries(presentation_tests PRIVATE pda_ui)
add_test(NAME presentation_science_finance_and_names COMMAND presentation_tests)

option(PDA_SANITIZERS "Use address/undefined behavior sanitizers on the host" ON)
if(PDA_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target pda_core pda_ui lvgl_host core_tests ui_tests organizer_tests priority_tests workspace_tests productivity_tests robustness_tests presentation_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(productivity_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(robustness_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(presentation_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(workspace_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(priority_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(core_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(ui_tests PRIVATE -fsanitize=address,undefined)
    target_link_options(organizer_tests PRIVATE -fsanitize=address,undefined)
endif()
